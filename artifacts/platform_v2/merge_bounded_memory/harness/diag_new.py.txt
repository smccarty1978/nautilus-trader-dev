"""Phase-1/4 step attribution for the NEW streamed nested merge (scratch output; read-only inputs).
usage: diag_new.py <worktree> <out_dir> <batch_rows|0> <partition_dir> [...]"""
import sys, json, time, threading
WT, OUT, BR = sys.argv[1], sys.argv[2], int(sys.argv[3]); PARTS = sys.argv[4:]
sys.path.insert(0, WT)
import psutil, pyarrow as pa, pyarrow.parquet as pq
from pathlib import Path
import research_workflow.lifecycle_v2 as lv2
proc = psutil.Process(); peak = [0]; stop = [False]
def sampler():
    while not stop[0]:
        r = proc.memory_info().rss
        if r > peak[0]: peak[0] = r
        time.sleep(0.02)
threading.Thread(target=sampler, daemon=True).start()
steps = []; t0 = time.time(); tl = [t0]
def mark(name, **kw):
    gib = lambda b: round(b / 1024**3, 3); now = time.time()
    steps.append({"step": name, "t_s": round(now - t0, 1), "dt_s": round(now - tl[0], 1), "rss_gib": gib(proc.memory_info().rss),
                  "peak_since_prev_gib": gib(peak[0]), "arrow_pool_now_gib": gib(pa.total_allocated_bytes()),
                  "arrow_pool_max_so_far_gib": gib(pa.default_memory_pool().max_memory()), **kw})
    tl[0] = now; peak[0] = proc.memory_info().rss; print(steps[-1], flush=True)
out = Path(OUT); out.mkdir(parents=True, exist_ok=True)
srcs = [Path(p) / "candidates.parquet" for p in PARTS]
mark("start", source_rows=sum(pq.ParquetFile(s).metadata.num_rows for s in srcs))
keys, order = lv2._merge_key_order(srcs); dup = bool(keys.duplicated(list(lv2.KEY)).any()); del keys
mark("_merge_key_order + duplicate check (KEY columns only)", dup=dup, identity_order=bool((order == __import__("numpy").arange(len(order))).all()))
st = lv2._stream_merged_parquet(out / "candidates.parquet", srcs, order, batch_rows=BR or None)
mark("_stream_merged_parquet (Arrow streaming write)", **st)
if not __import__("os").environ.get("DIAG_SKIP_IDENTITY"):
    idn = lv2._streaming_frame_content_identity(out / "candidates.parquet")
    mark("_streaming_frame_content_identity", identity=idn["identity"])
stop[0] = True
json.dump({"harness": "diag_new.py", "partitions": PARTS, "steps": steps}, open(out / "diag_new_steps.json", "w", encoding="utf-8"), indent=1)
