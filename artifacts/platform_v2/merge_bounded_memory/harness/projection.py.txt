"""Full-year projection from MEASURED final-code runs (pilot 1x, stress 8x). usage: projection.py <evidence_dir>"""
import json, os, sys
S = os.path.dirname(os.path.abspath(__file__)); E = sys.argv[1]
L = lambda *p: json.load(open(os.path.join(S, *p), encoding="utf-8"))
pts = {}
for lab, rows in (("final2_pilot", 13501), ("final2_stress8", 108008)):
    r = L("rss", f"{lab}.rss.json"); pts[lab] = {"rows": rows, "peak_tree_rss_gib": r["peak_tree_rss_gib"], "elapsed_s": r["elapsed_s"]}
(a_rows, a_peak, a_t), (b_rows, b_peak, b_t) = [(p["rows"], p["peak_tree_rss_gib"], p["elapsed_s"]) for p in pts.values()]
slope = (b_peak - a_peak) / (b_rows - a_rows); fixed = a_peak - slope * a_rows
tslope = (b_t - a_t) / (b_rows - a_rows); tfixed = a_t - tslope * a_rows
rows_per_day = 13501 / 11
year_rows = round(rows_per_day * 257)
old = L("rss", "diag_old.rss.json")["peak_tree_rss_gib"]
out = {
  "LABEL": "PROJECTION (not a measurement). Only the 13,501-row pilot and the 108,008-row merge-mechanics stress were run.",
  "measured_points_final_code": pts,
  "fit": {"peak_rss_gib": f"{fixed:.3f} + {slope*1e6:.3f}e-6 * rows", "elapsed_s": f"{tfixed:.1f} + {tslope*1e3:.3f}e-3 * rows",
          "note": "two-point linear fit; the per-row term covers O(rows) KEY/order/cost arrays (~48 B/row + pandas overhead) and allocator noise"},
  "full_year_rows_assumption": {"rows": year_rows, "basis": "pilot density 13,501 rows / 11 days x 257 Globex trading days (2022); profile width as in the pilot"},
  "projected_full_year": {"peak_rss_gib": round(fixed + slope * year_rows, 2), "elapsed_h": round((tfixed + tslope * year_rows) / 3600, 2)},
  "old_path_projection": {"peak_rss_gib_measured_pilot": old, "projected_full_year_gib": round(old * year_rows / 13501, 1), "basis": "every allocation on the old path is whole-partition (allocation_map.json)"},
  "caveats": [
    "row width (profile bins) of a real full year can exceed the 11-day pilot's; chunks are byte-bounded so this changes elapsed time, not the bound",
    "the bound also includes ONE source row group per partition: the chunked collection sink flushes at <=50,000 rows / 256 MiB estimated, so a partition row group can be ~256 MiB (decoded Arrow may differ)",
    "LifecycleV2.reconcile (the stage BEFORE merge) still pd.read_parquet's every whole partition: ~2.74 GiB per 13,501 G2 rows measured -> ~64 GiB projected for a full year; a full-year G2 lifecycle cannot reach MERGE until reconcile is bounded",
    "downstream readers of the merged file (fit _train_frame, controller_actions._merged_frames) load it whole; outside MERGE"
  ],
}
json.dump(out, open(os.path.join(E, "projection.json"), "w", encoding="utf-8"), indent=1); print(json.dumps(out["projected_full_year"]), out["fit"])
